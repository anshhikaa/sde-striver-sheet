//sort and two pointer 
class Solution {
public:
    int numRescueBoats(vector<int>& people, int limit) {
        sort(people.begin(),people.end());
        int light = 0 ; 
        int heavy = people.size()-1 ;
        int minBoats = 0 ;
        while(light <= heavy){
            if(people[light] + people[heavy] <= limit){
                minBoats++;
                light++;
                heavy--;
            }
            else {
                //send only heavy 
                minBoats ++;
                heavy--;
            }
        }
        return minBoats;
    }
};
