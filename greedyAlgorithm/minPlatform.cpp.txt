class Solution {
  public:
    int minPlatform(vector<int>& arr, vector<int>& dep) {
      int n = arr.size();
      int i = 0 ;
      int j = 0 ;
      int platformCount = 0 ;
      int ans = 0 ;
      sort(arr.begin(),arr.end());
      sort(dep.begin(),dep.end());
      while( i < n && j < n){
          if(arr[i] <= dep[j]){
              //train arrives 
              i++;
              platformCount ++;
              ans = max(platformCount ,ans);
          }
          else{
              //train departs 
              platformCount-- ;
              j ++;
              
          }
        
      }
      
      return ans ;  
    }
};
