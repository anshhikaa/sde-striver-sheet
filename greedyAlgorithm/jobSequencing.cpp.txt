class Solution {
  public:
    vector<int> jobSequencing(vector<int> &deadline, vector<int> &profit) {
        int n = profit.size();
        //create job 
        vector<pair<int,int>>job(n);
        int maxDeadline = -1;
        for(int i = 0 ; i < n ; i++) {
            job[i]= {profit[i],deadline[i]};
            maxDeadline = max(maxDeadline,deadline[i]);
        }
        //sort according desc profit
        sort(job.begin(),job.end(),[](pair<int,int>&a , pair<int,int>&b){
            return a.first > b.first;
        });
        vector<int>deadlines(maxDeadline+1,-1);
        int jobCount = 0 ;
        int totalProfit = 0;
        for(int i = 0 ; i < n ; i++){
            for(int j = job[i].second ; j > 0 ; j--){
                //if that deadline free(-1)then assign and break move to next job else try <deadline slots
                if(deadlines[j]== -1){
                    deadlines[j]= i ;
                    jobCount++;
                    totalProfit+=job[i].first;
                    break;
                }
            }
        }
        return{jobCount,totalProfit};
    }
};
