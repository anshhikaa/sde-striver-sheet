class Solution {
  public:
    vector<int> topKSumPairs(vector<int>& a, vector<int>& b, int k) {
        sort(a.begin(),a.end(),[](int &a1 ,int &a2){
            return a1 > a2;
        });
        sort(b.begin(),b.end(),[](int &b1 ,int &b2){
            return b1 > b2;
        });
        set<pair<int,int>>vis;
        vector<int>ans;
        priority_queue<pair<int,pair<int,int>>>maxHeap;
        maxHeap.push({a[0]+b[0],{0,0}});
        vis.insert({0,0});
        while(ans.size()< k){
            auto[i,j]= maxHeap.top().second;
            int sum = maxHeap.top().first;
            maxHeap.pop();
            ans.push_back(sum);
            //push one pair 
            if(j + 1 < b.size() && vis.find({i,j+1}) == vis.end()){
               maxHeap.push({a[i] + b[j+1], {i,j+1}});
                vis.insert({i,j+1});
            }

             // if those coordinates not found then push 
           if(i + 1 < a.size() && vis.find({i+1,j}) == vis.end()){
             maxHeap.push({a[i+1] + b[j], {i+1,j}});
             vis.insert({i+1,j});
             }
           
        }
      return ans ;  
    }
};
