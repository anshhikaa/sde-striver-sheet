//understanding the use of minheap as cooldown while maxheap as ready
class Solution {
public:
    int leastInterval(vector<char>& tasks, int n) {
        unordered_map<char,int>freq ;
        for(int i = 0 ; i < tasks.size() ; i++){
            freq[tasks[i]]++;
        }
        priority_queue<int>ready ;
        priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>>coolDown ;
        int interval = 0 ;
        for(auto it: freq){
            ready.push(it.second);
        }
        while(!ready.empty() || !coolDown.empty()){
            //check if cooldown top task is ready to be performed
           while(!coolDown.empty() && interval >= coolDown.top().first){
                ready.push(coolDown.top().second);
                coolDown.pop();
            }
            if(!ready.empty()){
                //max freq task performed
                int freq = ready.top(); ready.pop();
                interval++;
                //now goes into cooldown if left freq-1>1 
                if(freq > 1){
                    coolDown.push({interval+n,freq-1});
                }
            }
            else{
                //idle 
                interval++;
            }
           
        }
        return interval ;
    }
};
