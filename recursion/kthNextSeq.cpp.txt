//brute force : recursion - generate all permutatations -tc (n!* n * n!logn!)
class Solution {
public:
    void generate(int indx , string&nums,vector<string>&ans){
        //base case 
        if(indx == nums.length()){
            ans.push_back(nums);
            return ;
        }
        for(int i = indx ; i < nums.length() ; i++){
            swap(nums[indx], nums[i]);
            generate(indx+1,nums,ans);
            swap(nums[indx],nums[i]); // backtrack
        }
    }
    string getPermutation(int n, int k) {
        vector<string>ans ;
        string nums = "";
        for(int i = 1 ; i <= n ; i++){
            nums = nums + to_string(i);
        }
        generate(0,nums,ans);
        sort(ans.begin(),ans.end());
        return ans[k-1];
    }
};

//not with recursion - looping , finding relation- maths -tc n^2
class Solution {
public:
    string getPermutation(int n, int k) {
        int fact = 1 ;
        vector<int>num ;
        for(int i = 1 ; i < n ; i++){
            fact = fact * i ;
            num.push_back(i);
        }
        num.push_back(n);
        k = k-1  ; // oth based indexing
        string ans = "";
        while(true){
            ans = ans + to_string(num[k/fact]);
            num.erase(num.begin()+k/fact);
            if(num.size() == 0){
                break;
            }
            k = k % fact ;
            fact = fact / num.size();
        }
        return ans ;  
    }
};
