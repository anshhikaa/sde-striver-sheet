class Solution {
public:
    bool checkPalindrome(string s, int start , int end){
        while(start <= end){
            if(s[start] != s[end]){
                return false;
            }
            start ++;
            end--;
        }
        return true ;
    }
    void recursion(string s , int indx , vector<vector<string>>&ans,vector<string>&currPartition){
        //base case
        if( indx == s.length()){
            ans.push_back(currPartition);
            return ;
        }
        //loop 
        for(int i = indx ; i < s.length();i++){
          if(checkPalindrome(s,indx,i) == true){
            //partition 
            currPartition.push_back(s.substr(indx,i-indx +1));
            recursion(s,i+1,ans,currPartition);
            currPartition.pop_back();
          }
        }   
    }
    vector<vector<string>> partition(string s) {
        vector<vector<string>>ans ;
        vector<string>currPartition ;
        recursion(s,0,ans,currPartition);
        return ans ;
    }
};
