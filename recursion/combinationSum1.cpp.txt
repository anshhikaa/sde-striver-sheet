class Solution {
public:
    void findCombination(vector<int>& candidates, int target , int i , vector<int>&comb , vector<vector<int>>&ans){
        //base case 
        if(target == 0 ){
            ans.push_back(comb);
            return; 
        }
        if(target < 0 || i >= candidates.size()){
            return ;
        }
        //pick same
        comb.push_back(candidates[i]);
        findCombination(candidates,target-candidates[i],i,comb,ans); 
        // pick next
        comb.pop_back();
        findCombination(candidates,target,i+1,comb,ans); 
    }
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        vector<vector<int>>ans ;
        vector<int>comb;
        findCombination(candidates,target,0,comb,ans);
        return ans ;  
    }
};
