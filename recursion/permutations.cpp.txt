//generate all permutatations
class Solution {
public:
    void generate(int indx , vector<int>&nums,vector<vector<int>>&ans){
        //base case 
        if(indx == nums.size()){
            ans.push_back(nums);
            return ;
        }
        for(int i = indx ; i < nums.size() ; i++){
            swap(nums[indx], nums[i]);
            generate(indx+1,nums,ans);
            swap(nums[indx],nums[i]); // backtrack
        }
    }
    vector<vector<int>> permute(vector<int>& nums) {
        vector<vector<int>>ans ;
        generate(0,nums,ans);
        return ans ;
    }
};
