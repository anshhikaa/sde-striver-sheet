class Solution {
  public:
    int longestKSubstr(string &s, int k) {
        int left = 0 ;
        int right = 0 ;
        int maxLength = -1 ;
        int n = s.length();
        unordered_map<char,int>freq ;
        while(right < n){
            freq[s[right]]++;
            //if exceed then remove left until map size <= k
            while(freq.size()>k){
                freq[s[left]]--;
                if(freq[s[left]]==0){
                    freq.erase(s[left]);
                }
                left++;
            }
            if(freq.size() == k){
                maxLength = max(maxLength, right-left+1);
            }
            right ++;
        }
        return maxLength ;
        
    }
};
