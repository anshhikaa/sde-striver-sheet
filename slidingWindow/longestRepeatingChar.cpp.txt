class Solution {
public:
    int characterReplacement(string s, int k) {
        int left =  0;
        int right =  0;
        int maxLength = 0 ;
         int maxFreq = 0 ;
        unordered_map<char,int>freq ;
        while(right < s.length()){
            char c = s[right];
            freq[c]++;
            maxFreq = max(maxFreq,freq[c]);
            //validity
            //find operations 
            int maxOp = (right-left+1)- maxFreq ;
            while(maxOp > k){
                freq[s[left]]--;
                //maxFreq reset and update
                maxFreq = 0 ; 
                for(auto it: freq){
                    maxFreq = max(maxFreq,it.second);
                }
                left++;
                maxOp = (right-left+1) - maxFreq;
            }
            if(maxOp <= k){
                maxLength = max(maxLength,right-left+1);
            }
            right++;
        }
       return maxLength; 
    }
};
