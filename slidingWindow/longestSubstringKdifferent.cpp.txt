class Solution {
public:
  int count(vector<int>& nums, int k) {
        unordered_map<int,int>freq ;
        int left = 0 ;
        int right = 0 ;
        int count = 0;
        if( k < 0)return 0 ;
        while(right < nums.size()){
            freq[nums[right]]++;
            while(freq.size() > k){
              freq[nums[left]]--;
              if(freq[nums[left]] == 0){
                freq.erase(nums[left]);
              }
              left++;
            }
            if(freq.size() <= k){
                count += right-left+1;
            }
            right ++;
        }
      return count ;
    }
    int subarraysWithKDistinct(vector<int>& nums, int k) {
        return count(nums,k)-count(nums,k-1);
    }
};
