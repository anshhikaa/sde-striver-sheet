class Solution {
public:
int count(vector<int>&nums,int k){
    if(k < 0)return 0 ;
    int left = 0 ;
    int right = 0 ;
    int count = 0 ;
    int oddCount = 0 ;
    while(right < nums.size()){
        if(nums[right] % 2 == 1){
            oddCount++;
        }
        while(oddCount > k){
            if(nums[left] %2 == 1){
                oddCount--;
            }
            left++;
        }
           if(oddCount <= k){
            count += right-left+1;
        }
        right ++;
    }
    return count ;
}
    int numberOfSubarrays(vector<int>& nums, int k) {
        return count(nums,k)-count(nums,k-1);
    }
};
