class Solution {
public:
    int numberOfSubstrings(string s) {
        vector<int>freq(3,-1);
        int right = 0 ;
        int count= 0 ;
        while(right < s.length()){
            char c = s[right];
            freq[c-'a'] = right ;
            if(freq[0] != -1 && freq[1] != -1 && freq[2] != -1){
                int smallestIndex = min(min(freq[0],freq[1]),freq[2]);
                count += smallestIndex + 1 ;
            }
            right++;
        }
       return count ; 
    }
};
