class Solution {
public:
void dfs(vector<vector<int>>& isConnected,vector<bool>&vis, int v,int node){
      vis[node] = 1 ;
      for(int i = 0 ; i < v; i++){
        if(isConnected[node][i]== 1 && !vis[i]){
            dfs(isConnected,vis,v,i);
        }
      }
}
    int findCircleNum(vector<vector<int>>& isConnected) {
        int v= isConnected.size(); 
        vector<bool>vis(v,0);
        int count = 0 ;
        for(int i = 0 ; i < v ; i++){
            if(!vis[i]){
                dfs(isConnected,vis,v,i);
                count ++;
            }
        }
        return count ;
    }
};
