class MyStack {
public:
queue<int>q ;
    MyStack() {   
    }
    
    void push(int x) {
        int n = q.size();
        q.push(x);
        while(n--){
            //reversing the order so front of q acts axts as the top of stack
            q.push(q.front());
            q.pop();
        }
    }
    
    int pop() {
        int ans = q.front();
        q.pop();
        return ans ; 
    }
    
    int top() {
        return q.front();
    }
    
    bool empty() {
        return q.empty();
    }
};

 // using deque
 class MyStack {
public:
  deque<int>q ;
    MyStack() {}
    void push(int x) {
        q.push_back(x);
    }
    
    int pop() {
        int e = q.back();
        q.pop_back();
        return e ;  
    }
    
    int top() {
        int e = q.back();
        return e ;   
    }
    
    bool empty() {
       return q.empty(); 
    }

};
