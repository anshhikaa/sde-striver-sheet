//brute - o(n^2)
class Solution {
	public:
	vector<int>slidingWindowMin(vector<int>&arr, int windowSize) {
		vector<int>ans ;
		deque<int>dq ;
		for (int i = 0 ; i < arr.size() ; i++) {
			int element = arr[i];
			// maintain window size
			if (!dq.empty() && dq.front() <= i - windowSize) {
				dq.pop_front();
			}
			// maintain monotonic nature - >front = min ,then increasing order
			while (!dq.empty() && element <= arr[dq.back()]) {
				dq.pop_back();
			}
			dq.push_back(i);
			if (i >= windowSize - 1) {
				ans.push_back(arr[dq.front()]);
			}
		}
		
		return ans ;
	}
	vector<int> maxOfMins(vector<int>& arr) {
		vector<int>ans(arr.size()) ;
		for (int k = 1 ; k <= arr.size(); k++) {
			vector<int>min = slidingWindowMin(arr, k);
			int maxi = -1 ;
			for (int i = 0; i < min.size(); i++) {
				maxi = max(min[i], maxi);
			}
			ans[k - 1] = maxi;
		}
		return ans ;
	}
};

//optimized - o(n) - Histogram pattern
class Solution {
  public:
  vector<int> nextMin(vector<int>&arr){
      stack<int>q ;
      vector<int>next(arr.size()) ; 
      for(int i = arr.size()-1 ; i >= 0 ; i--){
          int element= arr[i];
          while(!q.empty() &&  element <= arr[q.top()]){
              q.pop();
          }
          next[i] = q.empty() ? arr.size() : q.top();
           q.push(i);
      }
      return next ;
      
  }
  vector<int> prevMin(vector<int>&arr){ 
        vector<int>prev(arr.size()) ;
        stack<int>q ;
        for(int i = 0 ; i < arr.size() ; i++){
            int element = arr[i];
            while(!q.empty() && element <= arr[q.top()]){
                q.pop();
            }
            prev[i] = q.empty() ? -1 : q.top();
            q.push(i);
        }
      return prev;  
      
  }
    vector<int> maxOfMins(vector<int>& arr) {
      vector<int>next  = nextMin(arr);
      vector<int>prev = prevMin(arr);
      vector<int>ans(arr.size()+1,0);
      for(int i = 0 ; i < arr.size() ; i++){
          int windowSize = next[i]-prev[i]-1 ;
          ans[windowSize]= max(ans[windowSize],arr[i]);
      }
      //fill misiing ans 
      for(int i = arr.size()-1 ; i >= 1 ; i-- ){
          ans[i]= max(ans[i],ans[i+1]);
      }
      ans.erase(ans.begin());
      return ans ;  
    }
};
