class Solution {
  public:
    int celebrity(vector<vector<int>>& mat) {
        int n = mat.size();
        vector<int>knownBy(n,0);
        vector<int>knows(n,0);
        for(int i  = 0 ; i < n ; i++){
            for(int j = 0 ; j < n; j ++){
                if(mat[i][j] == 1){
                    knows[i]++;
                    knownBy[j]++;
                }
            }
        }
        for(int i = 0 ; i < n  ; i++){
            if(knownBy[i] == n && knows[i] == 1 ){
                return i;
            }
        }
       return -1 ; 
    }
};

class Solution {
  public:
    int celebrity(vector<vector<int>>& mat) {
    int n = mat.size();
     int top  = 0 ;
     int down = n-1 ;
    while( top < down){
        if(mat[top][down] == 1){
            top++;
        }
        else if(mat[down][top]== 1){
            down--;
        }
        else{
            //neither top knows down nor down knows top both cant be celeb 
            top++;
            down--;
        }
    }
    if(top > down)return -1 ;
    for(int i = 0 ; i <n ; i++){
        if(top==i && mat[top][i] == 1){
              //knows themselves
                continue;
        }
        else if(mat[top][i]== 0 && mat[i][top]== 1){
            continue;
        }
        else{
            return -1 ;
        }
     }
    return top ;
    
    }
};
