class Solution {
  public:
    vector<int> nextSmallerEle(vector<int>& arr) {
        //  code here
        stack<int>st ;
        st.push(INT_MIN);
        vector<int>ans(arr.size());
        for(int i = arr.size()-1 ; i >= 0 ; i--){
            int num = arr[i];
            while(!st.empty() && num <= st.top()){
                st.pop();
            }
            ans[i] = st.top() == INT_MIN ? -1 : st.top();
            st.push(num);
        }
        return ans ;
    }
};
