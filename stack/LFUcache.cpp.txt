class Node{
    public:
    int key ;
    int val ;
    int cnt ;
    Node* prev ;
    Node * next  ; 
    Node(int _key,int _val){
        key = _key ;
        val = _val ;
        cnt = 1 ;
    }
};
class List{
    public:
    int size ;
    Node*head ;
    Node * tail ;
    List(){
        head = new Node(-1,-1);
        tail = new Node(-1,-1);
        head->next = tail ;
        tail->prev = head ;
        size = 0 ;
    }
    void add(Node* newNode){
        Node * temp = head->next ;
        newNode->next = temp ;
        newNode->prev = head ;
        head->next = newNode;
        temp->prev = newNode;
        size++;
    }
    void deleteNode(Node* delNode){
        Node * prevDel = delNode->prev ;
        Node * nextDel = delNode->next ;
        prevDel->next= nextDel ;
        nextDel->prev =prevDel ;
        size--;
    }
};
class LFUCache {
public:
    unordered_map<int,List*>freq;
    unordered_map<int,Node*>nodeAdd;
    int leastFreq ;
    int maxCap ;
    int currSize ;
    LFUCache(int capacity) {
        leastFreq = 0 ;
        maxCap = capacity ;
        currSize = 0 ;
    }
    void update(Node* node){
        freq[node->cnt]->deleteNode(node);//delete node from prev freq list
        List * newList = new List();
        // if last node in least freq removed then increase least freq  count -edge case
        if(node->cnt == leastFreq && freq[node->cnt]->size == 0 ){
            leastFreq++;
        }
        //already if that cnt+1 list exists 
        if(freq.find(node->cnt+1) != freq.end()){
            //initialise new list with exisitng 
            newList = freq[node->cnt+1];
        }
        //update node pos in cnt+1 list and node count increase
        node->cnt ++;
        newList->add(node);
        //modified list add freq map 
        freq[node->cnt]= newList ;
    }
    int get(int key) {
        //check existence 
        // if exist 
        if(nodeAdd.find(key) != nodeAdd.end()){
            Node * node = nodeAdd[key];
            int val = node->val ;
            //exits update its pos in freq map list , freq ++;
            update(node);
            return val;
            }
        // not exist
        return -1 ;
    }
    void put(int key, int value) {
        if(maxCap == 0)return ;
        //if exist 
        if(nodeAdd.find(key) != nodeAdd.end()){
            //update pos in freq+1 ;
            Node * node = nodeAdd[key];
            node->val = value ;
            update(node);
        }
        else{
            //new key-node not exist check capacity
            if(currSize == maxCap){
                //remove lfu 
                //check if lfu multiple 
                List* list = freq[leastFreq];
                nodeAdd.erase(list->tail->prev->key);
                list->deleteNode(list->tail->prev);
                currSize--;
            }
            currSize ++;
            //add new node at freq=1 
            leastFreq =1 ;
            List* list = new List();
            if(freq.find(leastFreq) != freq.end()){
                list = freq[leastFreq];
            }
            Node* node = new Node(key,value);
            list->add(node);
            nodeAdd[key] = node;
            freq[leastFreq]= list ;
        }
        
    }
};
