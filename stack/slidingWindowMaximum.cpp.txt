//use deque to push or pop from both side , and act as monotonic decreasing stack
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int>dq ;
        int n =nums.size();
        vector<int>ans ;
        for(int i= 0 ; i < n ; i++){
            //maintain window size remove from front if out of range
            if(!dq.empty() && dq.front() <= i-k){
                dq.pop_front();
            }
            // monotonic stack (decreasing order) front = maxElement, if any new element > back , to maintain monotonic nature pop till empty or larger element found
            while(!dq.empty() && nums[dq.back()] <= nums[i]){
                dq.pop_back();
            }
            //add current index
            dq.push_back(i);
            //store answer for ith iteration 
            if(i >= k-1){
             ans.push_back(nums[dq.front()]);
            } 
        }
        return ans ;
    }
};
