//for information about the popped elements store price,span of price 
class StockSpanner {
public:
    stack<pair<int,int>>st ;
    StockSpanner() {   
    }
    
    int next(int price) {
        //price , span
        int span = 1 ;
        while(!st.empty() && st.top().first <= price ){
            span += st.top().second; // add all prev spans
            st.pop(); 
        }
        st.push({price,span});
        return st.top().second;
    }
};

/**
 * Your StockSpanner object will be instantiated and called as such:
 * StockSpanner* obj = new StockSpanner();
 * int param_1 = obj->next(price);
 */
