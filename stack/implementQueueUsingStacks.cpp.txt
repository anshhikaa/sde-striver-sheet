class MyQueue {
public:
stack<int>stackIp;
stack<int>stackOp;
    MyQueue() {}
    
    void push(int x) {
        //push everything in stackip
        stackIp.push(x);   
    }
    
    int pop() {
        //push everything in stackIp to stack Op
        if(stackOp.empty()){
        while(!stackIp.empty()){
            stackOp.push(stackIp.top());
            stackIp.pop();
        }
        }
        int ans = stackOp.top();
        stackOp.pop();
        return ans ;
    }
    
    int peek() {
    if(stackOp.empty()){  
        while(!stackIp.empty()){
            stackOp.push(stackIp.top());
            stackIp.pop();
        }
    }
    return stackOp.top(); 
  }
    
    bool empty() {
        return stackIp.empty() && stackOp.empty();
    }
};

 * int param_3 = obj->peek();
