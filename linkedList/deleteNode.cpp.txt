/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    void deleteNode(ListNode* node) {
    // val of node->next 
    int value = node->next->val ;
    ListNode * temp = node->next->next ;
    node->val = value ;
    node->next = temp ;   
    }
};
