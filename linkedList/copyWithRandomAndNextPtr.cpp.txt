//brute force with hashmap + traversal
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        //create deep copy for each node in the given linked list 
        Node* temp = head ;
        unordered_map<Node*,Node*>map ;
        while(temp != nullptr){
            Node*deepcopy = new Node(temp->val);
            map[temp]= deepcopy ;
            temp = temp->next ;
        }
        //joining next and random pointers
        temp = head ;
        while(temp != nullptr){
            Node* nextNode = temp->next ;
            if(nextNode != nullptr){
                map[temp]->next = map[nextNode];
            }
            else{
                map[temp]->next = nullptr ;
            }
            Node * randomNode = temp->random;
            if(randomNode != nullptr){
                map[temp]->random = map[randomNode];
            }
            else{
                map[temp]->random = nullptr;
            }
            temp = temp->next ;
            
        }
        if(head == nullptr) return nullptr;
        return map[head];
    }
};

// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        //create copy nodes 
        Node * temp = head ;
        while(temp != nullptr){
            Node * newNode = new Node(temp->val);
            Node * forw = temp -> next ;
            temp->next = newNode  ;
            newNode->next = forw ;
            temp = forw ;
        }
        //point random pointers 
        temp = head ;
        while(temp != nullptr){
            //copy = temp->next  , actual next node =temp->next->next
            Node* copyNode = temp->next;
            if(temp->random != nullptr){
                Node * copyNodeRandom = temp->random->next;
                copyNode->random = copyNodeRandom ;
            }
            else{
                copyNode->random = nullptr ;
            }
            temp = temp->next->next; 
        }
        //point next pointers and seperate the lists
        Node* dummyNode = new Node(-1);
        temp = head ;
        Node* res = dummyNode ;
        while(temp!= nullptr){
        Node* copyNode = temp->next ;
          res->next = copyNode ;
          temp->next = copyNode->next ;
          res = copyNode;
          temp = temp->next ;
        }
     return dummyNode->next ;  
    }
};
