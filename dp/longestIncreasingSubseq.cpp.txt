//memoization solution
class Solution {
public:
int recursive(vector<int>&nums,int curr , int prev,vector<vector<int>>&dp){
    //base case 
    if(curr == nums.size()){
        return 0; 
    }
    //memoization
    if(dp[curr][prev+1] != -1)return dp[curr][prev+1];

    //pick only if prev index -1  or curr > prev
    int picklength=0 ; 
    if( prev == -1 || nums[curr] > nums[prev]){
        picklength =  1 + recursive(nums,curr+1,curr,dp);//curr becomes prev
    }
    //not pick - prev remains same
    int nonpicklength=  0 + recursive(nums,curr+1,prev,dp);
    return dp[curr][prev+1] = max(picklength,nonpicklength);
}
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>>dp(n,vector<int>(n+1,-1));
        return recursive(nums,0,-1,dp);
    }
};
//tabulation
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int>dp(n, 1);
        for(int curr = 0 ; curr < n; curr++){
            for(int prev= 0 ; prev < curr ; prev++){
                if(nums[prev] < nums[curr]){
                    dp[curr]= max(dp[curr],1+dp[prev]);
                }
            }
        }
        int maxLength = -1;
        for(int i = 0 ; i < n; i++){
          maxLength = max(maxLength,dp[i]);
        }
        return maxLength;
    }
};
//binary search 
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        vector<int>temp;
        temp.push_back(nums[0]);
        for(int i = 1 ; i < nums.size() ; i++){
            if(nums[i] > temp.back()){
                temp.push_back(nums[i]);
            }
            else{
                //lower_bound returns an iterator so to get index val substract temp.begin()
                int indx = lower_bound(temp.begin(),temp.end(),nums[i])-temp.begin();
                temp[indx]= nums[i];
            }
        }
       return temp.size(); 
    }
};
