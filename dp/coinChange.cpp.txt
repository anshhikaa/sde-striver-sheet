//recursion - tle
class Solution {
public:
int recursion(int i ,vector<int>&coins, int amount){
    if(i < coins.size() && amount == 0){
        return 0 ;
    }
    if(i >= coins.size() || amount < 0)return 1e9;
    // pick next 
    int nextPick = 0 + recursion(i+1,coins,amount);
    // pick itself (recursively)
    int pick = 1e9;
    if(coins[i]<= amount){
        pick = 1 + recursion(i,coins,amount-coins[i]);
    }
    return min(nextPick,pick);

}
    int coinChange(vector<int>& coins, int amount) {
        int result =  recursion(0,coins,amount);
        if(result == 1e9)return -1 ;
        return result ;
    }
};
// memoization 
class Solution {
public:
int recursion(int i ,vector<int>&coins, int amount,vector<vector<int>>&dp){
    if(i < coins.size() && amount == 0){
        return 0 ;
    }
    if(i >= coins.size() || amount < 0)return 1e9;
    if(dp[i][amount] != -1)return dp[i][amount];
    // pick next 
    int nextPick = 0 + recursion(i+1,coins,amount,dp);
    // pick itself (recursively)
    int pick = 1e9;
    if(coins[i]<= amount){
        pick = 1 + recursion(i,coins,amount-coins[i],dp);
    }
    return dp[i][amount] =min(nextPick,pick);
}
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        vector<vector<int>>dp(n,vector<int>(amount+1,-1));
        int result =  recursion(0,coins,amount,dp);
        if(result == 1e9)return -1 ;
        return result ;
    }
};
