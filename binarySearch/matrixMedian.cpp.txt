class Solution {
  public:
    int findUpperBound(vector<int>&arr, int target){
        int l = 0;
        int h = arr.size()-1;
        int ans  = arr.size() ;
        while(l <= h ){
            int mid = l + (h-l)/2;
            if(arr[mid] > target){
                ans = mid ;
                h = mid-1;
            }
            else{
               l = mid+1;
            }
        }
        return ans ;
    }
    int median(vector<vector<int>> &mat) {
        int n = mat.size();
        int m = mat[0].size();
        int reqMedian = ( n*m )/ 2; 
        //define search space 
        int l = INT_MAX;
        int h = INT_MIN ;
        for(int i = 0 ; i < n ; i++){
                l = min(l,mat[i][0]);
                h = max(h,mat[i][m-1]);
        }
        //binary search
        while(l <= h){
            int mid = l+(h-l)/2;
            //find no. of elements <= mid 
            int count = 0 ;
            for(int i = 0 ; i < n; i++){
                 count += findUpperBound(mat[i],mid);
                
            }
           
            if(count <= reqMedian){
                l = mid + 1 ;
            }
            else{
                h = mid - 1 ;
            }
        }
        return l ;
    }
};
