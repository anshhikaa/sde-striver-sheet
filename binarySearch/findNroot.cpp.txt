class Solution {
  public:
    long long findNsquare(int mid , int n){
        long long ans = 1 ;
        for(int i = 0 ; i < n ; i++){
            ans*= mid ;
            
        }
        return ans;
    }
    int nthRoot(int n, int m) {
        // Code here
        //define search space
        int l = 0 ;
        int h = m ;
        while(l <= h ){
            int mid = l+(h-l)/2 ;
            int nsquare = findNsquare(mid,n);
            if(nsquare == m){
                return mid ;
            }
            else if(nsquare > m){
                //left side 
                h = mid-1 ;
            }
            else{
                l = mid+1 ;
            }
            
        }
        return -1 ;
        
    }
};
