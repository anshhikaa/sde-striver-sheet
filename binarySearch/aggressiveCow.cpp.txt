//binary search on the answer min to max-min 
class Solution {
	public:
	bool canWePlace(int minDist, int cows, vector<int>&stalls) {
		int last = stalls[0]; // first cow always placed at the beginning
		int cowPlaced = 1 ;
		for (int i = 1 ; i < stalls.size() ; i ++) {
			if (stalls[i]-last >= minDist) {
				cowPlaced ++;
				last = stalls[i];
			}
		}
		if (cows <= cowPlaced)return true ;
		return false;
	}
	int aggressiveCows(vector<int> &stalls, int k) {
		sort(stalls.begin(), stalls.end());
		int n = stalls.size() - 1;
		int l = 1;
		int h = stalls[n];
		while (l <= h) {
			int mid = l + (h - l)/2 ;
			if (canWePlace(mid, k, stalls) == true) {
				// mid is smaller or equal to minDsit so explore larger values
				// explore right
				l = mid + 1;
			}
			else {
				// mid is greater than minDist so explore smaller values
				// explore left
				h = mid - 1 ;
				
			}
		}
		return h;
	}
};
