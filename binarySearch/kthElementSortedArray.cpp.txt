//same approach as median of sorted array but keep the edges case for l and h initialization ( k < n1 for h & k > n2 for l)
class Solution {
  public:
    int kthElement(vector<int> &a, vector<int> &b, int k) {
        int n1 = a.size();
        int n2 = b.size();
        if(n1 > n2) {
            return kthElement(b,a,k);
        }
        int l = max(0,k-n2) ;
        int h = min(k,n1) ;
        while( l <= h){
            int cut1  = l + (h-l)/2 ;
            int cut2 = k-cut1 ;
            int l1 = cut1 == 0 ? INT_MIN : a[cut1-1];
            int l2 = cut2 == 0 ? INT_MIN : b[cut2-1];
            int r1 = cut1 == n1 ? INT_MAX : a[cut1];
            int r2 = cut2 == n2 ? INT_MAX : b[cut2];
            if(l1 <= r2 && l2 <= r1){
                return max(l1,l2);
            }
            else if( l1 > r2){
                //move left
                h = cut1-1;
            }
            else{
                // r1 < l2
                //move right 
                l = cut1+1;
            }
        }
        return -1; 
    }
};
