/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int>preorder ;
        TreeNode* curr = root ;
        while(curr != nullptr){
            if(curr->left == nullptr){
               preorder.push_back(curr->val);
                curr = curr->right;
            }
            else{
                TreeNode * lastNode = curr->left;
                while(lastNode->right && lastNode->right != curr){
                    lastNode = lastNode->right;
                }
                if(lastNode->right == nullptr){
                    //make connection to curr
                    lastNode->right = curr;
                    preorder.push_back(curr->val);
                    curr= curr->left;
                }
                else{
                    //already connected then remove connection and move right
                    lastNode->right = nullptr ;
                    curr = curr->right;
                }
            }
        }
        return preorder;
    }
};
