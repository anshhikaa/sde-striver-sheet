class Solution {
  public:
    vector<int> topView(Node *root) {
         vector<int>topView;
        if(root == nullptr){
            return topView;
        }
       queue<pair<int,Node*>>q; // line,val
       map<int,int>mapping;  // line->1st node val
       q.push({0,root});
       while(!q.empty()){
           int line = q.front().first;
           Node* node = q.front().second ;
           q.pop();
           if(node->left){
                 q.push({line-1,node->left});
               
           }
           if(node->right){
           q.push({line+1,node->right});
           }
           if(mapping.find(line) == mapping.end()){
               mapping[line]= node->data;
           }
       }
       //map to vector 
       for(auto it: mapping){
           topView.push_back(it.second);
       }
       return topView;
    }
};
