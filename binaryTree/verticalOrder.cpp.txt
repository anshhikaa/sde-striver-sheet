/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> verticalTraversal(TreeNode* root) {
        map<int,map<int,multiset<int>>>mapping ; // line->every level->multiple nodes
        queue<pair<TreeNode*,pair<int,int>>>q; // {node,{level,line}}
        q.push({root,{0,0}});
        while(!q.empty()){
            TreeNode* node = q.front().first;
            int level = q.front().second.first;
            int line = q.front().second.second;
            q.pop();
            //map 
            mapping[line][level].insert(node->val);
            if(node->left){
            q.push({node->left,{level+1,line-1}});
            }
            if(node->right){
            q.push({node->right,{level+1,line+1}});
            }
        }
        vector<vector<int>>verticalOrder;
        for(auto it : mapping ){      // line 0 - level 0 - {1}, level 2 - {5,6}
            auto &levelIt = it.second; //    lineIt = level 0 then level 2
            vector<int>verticalLine;
            for(auto nodes :levelIt ){  // nodes - {0} then {5,6}
                auto &mset = nodes.second ;
                 verticalLine.insert(verticalLine.end(),mset.begin(),mset.end()); // {0}
            }
            verticalOrder.push_back(verticalLine);
        }
       return verticalOrder; 
    }
};
