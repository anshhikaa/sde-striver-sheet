class Solution {
  public:
  void traversal(Node* root , vector<int>&leftView,int level){
      //base condition 
      if(root==nullptr){
          return ;
      }
      if(level == leftView.size()){
          leftView.push_back(root->data);
      }
      traversal(root->left,leftView,level+1);
      traversal(root->right,leftView,level+1);
  }
    vector<int> leftView(Node *root) {
       vector<int>leftView;
       traversal(root,leftView,0);
       return leftView;
    }
};
