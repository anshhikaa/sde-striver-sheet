class Solution {
public:
int calculateSum(TreeNode* node , int &sum){
    //base case
    if(node == nullptr){
        return 0 ;
    }
    //taking max(0,)as we dont want to extend any negative val in stead return 0 
    int leftContribution = max(0,calculateSum(node->left,sum));
    int rightContribution = max(0 , calculateSum(node->right,sum));
    //what is the maxSum if curr node at highest point 
    sum = max(sum,node->val+leftContribution+rightContribution);
    //what is the best single path branch that i can extend further
    return node->val+max(leftContribution,rightContribution);
}
    int maxPathSum(TreeNode* root) {
        int sum  = INT_MIN;
        int pathSum = calculateSum(root,sum);
        return sum; 
    }
};
