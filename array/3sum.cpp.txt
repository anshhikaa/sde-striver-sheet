//tc = o(n^3) 
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
      int n = nums.size();
      set<vector<int>>st ;
      for(int i = 0 ; i < n ; i++){
        for(int j = i+1 ; j < n ; j++){
            for(int k = j+1 ; k <n ; k++){
                if(nums[i]+nums[j]+nums[k]== 0){
                    vector<int>temp = {nums[i],nums[j],nums[k]};
                    sort(temp.begin(),temp.end());
                    st.insert(temp);
                }
            }
        }
      }
      vector<vector<int>>ans(st.begin(),st.end());
      return ans ;    
    }
};

//tc= o(n^2) nested 2 loops + hashing
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
      int n = nums.size();
      set<vector<int>>st ;
      for(int i = 0 ; i < n ; i++){
        //reset hashset 
            set<int>hashset;
        for(int j = i+1 ; j < n ; j++){
                int req = -(nums[i]+nums[j]);
                if(hashset.find(req) != hashset.end()){
                   vector<int>temp = {nums[i],nums[j], req};
                    sort(temp.begin(),temp.end());
                    st.insert(temp);
                }
              else{
                    //empty or didnt find 
                    hashset.insert(nums[j]);
              }     
        }
     }
      vector<vector<int>>ans(st.begin(),st.end());
      return ans ;    
    }
};

//optimal - sort then 3 pointers
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>>ans ;
        int n = nums.size();
        sort(nums.begin(),nums.end());
        int i = 0 ;
        while(i < n ){
            int j = i+1 ;
            int k = n-1 ;
            while(j < k){
                // if answer
                if(nums[i] + nums[j]+nums[k] == 0 ){
                    ans.push_back({nums[i],nums[j],nums[k]});
                    int lastj = nums[j];
                    int lastk = nums[k];
                    while( j < k && nums[j] == lastj ){
                        j++;
                    }
                    while( k > j && nums[k] == lastk ){
                        k--;
                    }
                }
                //if less sum is < 0 then increase nums val
                else if(nums[i] + nums[j]+nums[k] < 0){
                    j++;
                }
                else{
                    //move than 0 then need to reduce val 
                    k--;
                }
            }
            //if j crosses k
            int lasti = nums[i];
            while(i < n && nums[i] == lasti ){
                i++;
            }
        }
       return ans ; 
    }
};
