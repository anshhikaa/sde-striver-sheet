// 3 pointer mid traversing , 4 parts ,0-low,low-mid,mid-high,high-end
class Solution {
public:
    void sortColors(vector<int>& nums) {
        int n = nums.size()-1;
        int low = 0 ;
        int mid = 0 ;
        int high =n ;
        while(mid <= high){
            if(nums[mid]==2){
                swap(nums[mid],nums[high]);
                high--;
            }
            else if(nums[mid]==0){
                swap(nums[mid],nums[low]);
                mid ++;
                low ++;
            }
            else{
                //mid == 1 
                mid ++;
            }
        } 
    }
};
