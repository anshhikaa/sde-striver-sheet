//better than brute approach , tc = n^2 , sc = m+n
class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int m = matrix.size();
        int n = matrix[0].size();        
        vector<int>row(m,0);
        vector<int>col(n,0);
        for(int i = 0 ; i < m ; i++){
            for(int j = 0 ; j < n ; j++){
                if(matrix[i][j]==0){
                    row[i]= 1 ;
                    col[j] = 1 ;
                }
            }
        }

        for(int i = 0 ; i < m ; i++ ){
            for(int j = 0 ; j <  n ; j++){
                if(row[i]|| col[j]){
                    matrix[i][j]= 0;
                }
            }
        }
    }
};

// final optimized solution - In-place modification using existing structure as marker storage
class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int col0 = 1 ;
        int n = matrix.size();
        int m = matrix[0].size();
        //row[]= matrix[i][0] , col[]= matrix[0][j]
        for(int i = 0 ; i < n ; i++){
            for(int j = 0 ; j  < m ; j++){
                if(matrix[i][j]==0){
                    //col mark
                    if(j != 0){
                     matrix[0][j]=0 ;
                    }
                    else{
                        col0=  0 ;
                    }
                    //row mark
                    matrix[i][0]= 0 ;
                }
            }
        }
        for(int i = 1 ; i < n ; i++){
            for(int j = 1 ; j < m ; j++){
                if(matrix[i][0] == 0||matrix[0][j]==0){
                    matrix[i][j] = 0;
                }
            }
        }
        //covering the 0th row and 0th col  
        if(matrix[0][0]==0){
            for(int i = 0 ; i < m ; i++){
                matrix[0][i]= 0 ;
            }
        }
        if(col0 == 0){
            for(int j = 0 ; j < n ; j++){
                matrix[j][0]=0;
            }
        }
    }
};
